# bob BRAM for yosys memory_libmap (M8) - see yosys passes/memory/memlib.md.
# Patterned on share/yosys/xilinx/brams_xc4v.txt's TDP block, reduced to bob's
# BRAM block (hw/src/tiles/bram_core.v, UG473 RAMB18E1 behaviour): 1024 x 18,
# one clock, EN gates reads and writes, RST resets the output latch only when
# EN (gated by the clock enable), output latch powers up 0, no output register
# used by synthesis, write mode per port.

ram block $__BOB_BRAM_ {
    abits 10;
    widths 18 global;
    cost 64;
    init any;
    port srsw "A" "B" {
        clock posedge "C";
        clken;
        rdsrst zero gated_clken;
        rdinit zero;
        portoption "WRITE_MODE" "WRITE_FIRST" {
            rdwr new;
        }
        portoption "WRITE_MODE" "READ_FIRST" {
            rdwr old;
        }
        portoption "WRITE_MODE" "NO_CHANGE" {
            rdwr no_change;
        }
        optional;
    }
}
